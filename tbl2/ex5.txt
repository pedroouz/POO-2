using System;

public class EstoqueInsuficienteException : Exception
{
    public EstoqueInsuficienteException(string mensagem)
        : base(mensagem)
    {
    }
}

public class Estoque
{
    public int Quantidade { get; private set; }

    public Estoque(int quantidadeInicial)
    {
        Quantidade = quantidadeInicial;
    }

    public void Retirar(int quantidade)
    {
        if (quantidade > Quantidade)
        {
            throw new EstoqueInsuficienteException(
                "Estoque insuficiente. Quantidade disponível: " + Quantidade
            );
        }

        Quantidade -= quantidade;
        Console.WriteLine("Produto retirado com sucesso!");
    }
}

class Program
{
    static void Main()
    {
        try
        {
            Estoque estoque = new Estoque(10);

            Console.Write("Digite a quantidade que deseja retirar: ");
            int quantidade = int.Parse(Console.ReadLine());

            estoque.Retirar(quantidade);

            Console.WriteLine("Quantidade restante: " + estoque.Quantidade);
        }
        catch (EstoqueInsuficienteException ex)
        {
            Console.WriteLine("Erro: " + ex.Message);
        }
        catch (FormatException)
        {
            Console.WriteLine("Digite uma quantidade válida.");
        }
    }
}
